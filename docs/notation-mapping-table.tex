% Tabla 2: Ecuaciones principales
\begin{table}[h]
\centering
\caption{Ecuaciones Principales: Antigua vs. Nueva Notación}
\label{tab:equations-mapping}

\begin{tabular}{|l|l|}
\hline
\textbf{Anterior} & \textbf{Actual} \\
\hline
\multicolumn{2}{|c|}{\textbf{Demanda}} \\
\hline
$\sum_{m} Z_{m,t,c} \ge Demanda_{t,c}$ &
$\sum_{m \in M} z_{m,t,c} \geq d_{t,c}$ \\
\hline
\multicolumn{2}{|c|}{\textbf{Consumo (Menú $\to$ Insumo)}} \\
\hline
$Consumo_{i,t} = \sum_c \sum_m \hat{R}_{i,m} \cdot Z_{m,t,c}$ &
$\gamma_{i,t} = \sum_{c \in C} \sum_{m \in M} a_{i,m} \cdot z_{m,t,c}$ \\
\hline
\multicolumn{2}{|c|}{\textbf{Lotes}} \\
\hline
$X_{i,t} = Lote_i \cdot Y_{i,t}$ &
$x_{i,t} = \ell_i \cdot y_{i,t}$ \\
\hline
\multicolumn{2}{|c|}{\textbf{Presupuesto}} \\
\hline
$\sum_t \sum_i C_i \cdot X_{i,t} \le \mathcal{B}$ &
$\sum_{t \in T} \sum_{i \in I} c_i \cdot x_{i,t} \leq B$ \\
\hline
\multicolumn{2}{|c|}{\textbf{Balance de Inventario}} \\
\hline
$Inv_{i,t} = Inv_{i,t-1} + X_{i,t} + Donaciones_{i,t} - Consumo_{i,t}$ &
$\mu_{i,t} = \mu_{i,t-1} + x_{i,t} + \delta_{i,t} - \gamma_{i,t}$ \\
\hline
\multicolumn{2}{|c|}{\textbf{Función Objetivo}} \\
\hline
$\min \mathcal{Z} = \sum_t \sum_i C_i \cdot X_{i,t}$ &
$\min \Phi = \sum_{t \in T} \sum_{i \in I} c_i \cdot x_{i,t}$ \\
\hline
\multicolumn{2}{|c|}{\textbf{Inventario Inicial}} \\
\hline
$Inv_{i,0} = \mathcal{I}_{i,0}$ &
$\mu_{i,0} = I_{i,0}$ \\
\hline
\multicolumn{2}{|c|}{\textbf{Condición Terminal}} \\
\hline
$Inv_{i,7} \ge \mathcal{I}_{i,0}$ &
$\mu_{i,7} \geq I_{i,0}$ \\
\hline
\end{tabular}
\end{table}
